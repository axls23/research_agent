class WorkQueue:
    def __init__(self, db, workers, retry_limit=3):
        self.db = db
        self.workers = workers
        self.retry_limit = retry_limit
        self.inflight = {}

    async def enqueue(self, job):
        await self.db.insert(job)

    async def lease(self, worker_id):
        job = await self.db.next_pending()
        if job is None:
            return None
        self.inflight[job.id] = worker_id
        await self.db.mark_running(job.id, worker_id)
        return job

    async def complete(self, job_id):
        self.inflight.pop(job_id, None)
        await self.db.mark_done(job_id)

    async def fail(self, job_id, error):
        self.inflight.pop(job_id, None)
        job = await self.db.get(job_id)
        if job.attempts >= self.retry_limit:
            await self.db.mark_dead(job_id, error)
        else:
            await self.db.mark_pending(job_id, error)
```
