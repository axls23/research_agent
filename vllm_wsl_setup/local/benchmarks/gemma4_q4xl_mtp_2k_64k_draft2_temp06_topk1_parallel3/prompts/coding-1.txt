class Scheduler:
    async def claim(self, worker):
        task = await self.db.fetch_one("select * from tasks where state='queued' order by priority desc limit 1")
        if not task:
            return None
        await self.db.execute("update tasks set state='running', worker=$1 where id=$2", worker, task["id"])
        return task

    async def finish(self, task_id):
        await self.db.execute("update tasks set state='done' where id=$1", task_id)

    async def fail(self, task_id, error):
        await self.db.execute("update tasks set attempts=attempts+1, state='queued', last_error=$2 where id=$1", task_id, error)
```
